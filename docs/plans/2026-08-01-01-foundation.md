# Фундамент: скелет репозитория, канон и валидаторы — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Собрать скелет репозитория по `docs/README.md` и реализовать канонические
константы данных плюс четыре уровня валидации из `docs/DATA_CONTRACT.md` — чистыми
функциями, без сети и без GPU.

**Architecture:** `contracts/` держит машиночитаемое описание канонической формы
(сетка, уровни, переменные, единицы, физические диапазоны) и не зависит ни от чего,
кроме numpy. `validators/` состоит из функций `xarray.Dataset -> list[Check]`,
сгруппированных по четырём уровням `TESTING.md`; агрегатор `validate()` собирает из них
`ValidationReport`, который сериализуется в `validation.json` рядом с артефактом.
Ни один валидатор не чинит данные и не пишет на диск — только выносит вердикт.

**Tech Stack:** Python 3.12, xarray, numpy, zarr>=3, pytest, ruff, mypy, pre-commit.

## Global Constraints

- Python 3.11–3.12; целевая версия CI — 3.12 (`docs/SETUP.md` §4).
- `zarr >= 3` — формат v3, проверять что версия действительно 3.x.
- Все времена внутри системы — UTC, `datetime64[ns]`, без таймзонных суффиксов
  (`docs/DOMAIN.md` §2, `docs/DATA_CONTRACT.md` §1).
- Единицы внутри — СИ. Перевод в °C/гПа/мм — только на слое API (`docs/DOMAIN.md` §4).
- Сетка 0.25°: широта `+90 → −90` строго убывает, 721 значение; долгота
  `−180 → 179.75` строго возрастает, 1440 значений (`docs/DATA_CONTRACT.md` §1).
- Уровни давления, ровно и в этом порядке: 50, 100, 150, 200, 250, 300, 400, 500,
  600, 700, 850, 925, 1000 гПа (`docs/DOMAIN.md` §5).
- Имена переменных — как у Aurora: `2t`, `10u`, `10v`, `msl`, `t`, `u`, `v`, `q`, `z`;
  статика `lsm`, `z`, `slt`. Всего 69 полей на шаг (`docs/DOMAIN.md` §5).
- Пропуски — только `NaN`, атрибут `_FillValue` проставлен; `-9999` и подобные
  запрещены (`docs/DATA_CONTRACT.md` §1).
- Отказ валидатора обязан называть **поле, полученное значение и ожидаемое**.
  Формулировка «validation failed» без деталей не принимается (`docs/DATA_CONTRACT.md` §4).
- Ни один тест не ходит в сеть (`docs/TESTING.md`).
- Direction of dependencies: `api → storage/cache`. `api/` не импортирует `adapters/`,
  `pipeline/`, `torch` (`docs/ARCHITECTURE.md`).

```python
def test_failure_message_names_field_value_and_expectation() -> None:
```

```python
"""Результат проверки. Отказ обязан называть поле, значение и ожидание.

docs/DATA_CONTRACT.md §4: «Формулировка "validation failed" без деталей
не принимается на ревью». Поэтому конструктор отказа принимает field/got/expected
по отдельности и собирает сообщение сам — забыть их нельзя.
"""

from dataclasses import dataclass, field as dc_field
from typing import Any

LEVELS = ("structure", "semantics", "physics", "sanity")


@dataclass(frozen=True, slots=True)
class Check:
    name: str
    level: str
    passed: bool
    message: str = ""
    details: dict[str, Any] = dc_field(default_factory=dict)

    def __post_init__(self) -> None:
        if self.level not in LEVELS:
            raise ValueError(f"level: got {self.level!r}, expected one of {LEVELS}")

    def to_dict(self) -> dict[str, Any]:
        return {
            "name": self.name,
            "level": self.level,
            "passed": self.passed,
            "message": self.message,
            "details": _jsonable(self.details),
        }


def ok(name: str, level: str, message: str = "", **details: Any) -> Check:
    return Check(name=name, level=level, passed=True, message=message, details=details)


def fail(name: str, level: str, field: str, got: Any, expected: Any) -> Check:
    return Check(
        name=name,
        level=level,
        passed=False,
        message=f"{field}: got {got!r}, expected {expected!r}",
        details={"field": field, "got": got, "expected": expected},
    )


@dataclass(frozen=True, slots=True)
class ValidationReport:
    checks: tuple[Check, ...]

    @property
    def ok(self) -> bool:
        return all(c.passed for c in self.checks)

    def failures(self) -> tuple[Check, ...]:
        return tuple(c for c in self.checks if not c.passed)

    def to_dict(self) -> dict[str, Any]:
        return {"ok": self.ok, "checks": [c.to_dict() for c in self.checks]}


def _jsonable(value: Any) -> Any:
    """numpy-скаляры не сериализуются json'ом; отказ обязан долетать до файла."""
    import numpy as np

    if isinstance(value, dict):
        return {k: _jsonable(v) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [_jsonable(v) for v in value]
    if isinstance(value, np.generic):
        return value.item()
    if isinstance(value, np.ndarray):
        return value.tolist()
    return value
```

```python
"""Синтетические канонические Dataset'ы — только для тестов самих валидаторов.

docs/TESTING.md запрещает синтетику там, где проверяется чтение источника:
у неё правильные оси и единицы, поэтому ловушки она не воспроизводит. Здесь
проверяется сам валидатор, и «правильные оси» — это ровно то, что нужно:
тест портит одну вещь за раз и смотрит, поймана ли она.
"""

from typing import Iterable

import numpy as np
import xarray as xr

from contracts import canon


def varying_field(shape: tuple[int, ...]) -> np.ndarray:
    """Поле с градиентом: константное поле валится проверкой «здравый смысл»."""
    total = int(np.prod(shape))
    return np.linspace(0.0, 1.0, total, dtype=np.float32).reshape(shape)


def canonical_dataset(
    times: int = 1,
    init_time: str = "2026-08-01T00:00:00",
    constant: float = 288.0,
    surface_vars: Iterable[str] | None = None,
    atmos_vars: Iterable[str] | None = None,
) -> xr.Dataset:
    surface = tuple(surface_vars) if surface_vars is not None else canon.SURFACE_VARS
    atmos = tuple(atmos_vars) if atmos_vars is not None else canon.ATMOS_VARS

    time = np.array(
        [np.datetime64(init_time) + np.timedelta64(6 * i, "h") for i in range(times)],
        dtype="datetime64[ns]",
    )
    ny, nx = canon.GRID_SHAPE
    nl = len(canon.PRESSURE_LEVELS)

    def field(shape: tuple[int, ...], value: float) -> np.ndarray:
        return np.broadcast_to(np.float32(value), shape)

    defaults = {"2t": constant, "10u": 3.0, "10v": -2.0, "msl": 101_325.0}
    data: dict[str, xr.DataArray] = {}
    for name in surface:
        arr = field((times, ny, nx), defaults.get(name, 1.0))
        data[name] = xr.DataArray(arr, dims=("time", "lat", "lon"))

    level_defaults = {"t": 250.0, "u": 10.0, "v": 5.0, "q": 0.004, "z": 50_000.0}
    for name in atmos:
        arr = field((times, nl, ny, nx), level_defaults.get(name, 1.0))
        data[name] = xr.DataArray(arr, dims=("time", "level", "lat", "lon"))

    ds = xr.Dataset(
        data,
        coords={
            "time": time,
            "level": np.array(canon.PRESSURE_LEVELS, dtype="int32"),
            "lat": canon.LAT,
            "lon": canon.LON,
        },
        attrs={
            "source": "ifs-analysis",
            "source_url": "test://fixture",
            "retrieved_at": "2026-08-01T07:41:12Z",
            "init_time": f"{init_time}Z",
            "kind": "analysis",
            "grid": "0.25deg-global",
            "adapter_version": "0.0.0-test",
        },
    )
    for name, var in ds.data_vars.items():
        var.attrs["units"] = canon.UNITS[str(name)]
        var.attrs["_FillValue"] = np.float32(np.nan)
    return ds
```

```python
"""Уровень «структура»: размерности, монотонность осей, полнота набора полей."""

import numpy as np
import xarray as xr

from contracts import canon
from validators.result import Check, fail, ok

LEVEL = "structure"


def check_structure(ds: xr.Dataset, *, expect_static: bool = False) -> list[Check]:
    checks: list[Check] = []
    checks.append(_grid_shape(ds))
    checks.append(_lat_descending(ds))
    checks.append(_lon_range(ds))
    checks.append(_lon_ascending(ds))
    checks.append(_levels_match(ds))
    checks.append(_fields_present(ds, expect_static=expect_static))
    if "time" in ds.coords:
        checks.append(_time_unique(ds))
        checks.append(_time_regular(ds))
    return checks


def _grid_shape(ds: xr.Dataset) -> Check:
    got = (ds.sizes.get("lat"), ds.sizes.get("lon"))
    if got != canon.GRID_SHAPE:
        return fail("grid_shape", LEVEL, "(lat, lon)", got, canon.GRID_SHAPE)
    return ok("grid_shape", LEVEL)


def _lat_descending(ds: xr.Dataset) -> Check:
    lat = np.asarray(ds["lat"].values)
    if lat.size < 2 or not np.all(np.diff(lat) < 0):
        return fail("lat_descending", LEVEL, "lat", f"{lat[:3]}...{lat[-3:]}", "strictly decreasing +90..-90")
    return ok("lat_descending", LEVEL)


def _lon_range(ds: xr.Dataset) -> Check:
    lon = np.asarray(ds["lon"].values)
    got = (float(lon.min()), float(lon.max()))
    expected = (-180.0, 179.75)
    if not np.isclose(got[0], expected[0]) or not np.isclose(got[1], expected[1]):
        return fail("lon_range", LEVEL, "lon", got, expected)
    return ok("lon_range", LEVEL)


def _lon_ascending(ds: xr.Dataset) -> Check:
    lon = np.asarray(ds["lon"].values)
    if lon.size < 2 or not np.all(np.diff(lon) > 0):
        return fail("lon_ascending", LEVEL, "lon", f"{lon[:3]}...{lon[-3:]}", "strictly increasing")
    return ok("lon_ascending", LEVEL)


def _levels_match(ds: xr.Dataset) -> Check:
    if "level" not in ds.coords:
        return ok("levels_match", LEVEL, "surface-only dataset")
    got = tuple(int(v) for v in np.asarray(ds["level"].values))
    if got != canon.PRESSURE_LEVELS:
        return fail("levels_match", LEVEL, "level", got, canon.PRESSURE_LEVELS)
    return ok("levels_match", LEVEL)


def _fields_present(ds: xr.Dataset, *, expect_static: bool) -> Check:
    expected = set(canon.SURFACE_VARS) | set(canon.ATMOS_VARS)
    if expect_static:
        expected |= set(canon.STATIC_VARS)
    missing = sorted(expected - set(map(str, ds.data_vars)))
    if missing:
        return fail("fields_present", LEVEL, "data_vars", f"missing {missing}", sorted(expected))
    return ok("fields_present", LEVEL, f"{canon.FIELDS_PER_STEP} fields per step")


def _time_unique(ds: xr.Dataset) -> Check:
    times = np.asarray(ds["time"].values)
    if times.size != np.unique(times).size:
        duplicated = [str(t) for t in times[:-1][times[1:] == times[:-1]]]
        return fail("time_unique", LEVEL, "time", f"duplicates {duplicated}", "unique timestamps")
    return ok("time_unique", LEVEL)


def _time_regular(ds: xr.Dataset) -> Check:
    times = np.asarray(ds["time"].values)
    if times.size < 2:
        return ok("time_regular", LEVEL, "single step")
    deltas = np.unique(np.diff(times).astype("timedelta64[h]").astype(int))
    if deltas.tolist() != [canon.STEP_HOURS]:
        return fail("time_regular", LEVEL, "time", f"steps of {deltas.tolist()} h", f"[{canon.STEP_HOURS}] h")
    return ok("time_regular", LEVEL)
```

```python
"""Уровень «семантика»: единицы, времена, уровни. Ловушки 1 и 3 из docs/DOMAIN.md §6."""

import numpy as np
import xarray as xr

from contracts import canon
from validators.result import Check, fail, ok

LEVEL = "semantics"


def check_semantics(ds: xr.Dataset) -> list[Check]:
    checks: list[Check] = []
    checks.extend(_units(ds))
    if "time" in ds.coords:
        checks.append(_time_naive_utc(ds))
        checks.append(_valid_time_consistent(ds))
    return checks


def _units(ds: xr.Dataset) -> list[Check]:
    checks: list[Check] = []
    for raw_name in ds.data_vars:
        name = str(raw_name)
        expected = canon.UNITS.get(name)
        if expected is None:
            checks.append(fail("units_known", LEVEL, name, "unknown variable", sorted(canon.UNITS)))
            continue
        got = ds[name].attrs.get("units")
        if got != expected:
            checks.append(fail("units", LEVEL, name, got, expected))
        else:
            checks.append(ok("units", LEVEL, f"{name} in {expected}"))
    return checks


def _time_naive_utc(ds: xr.Dataset) -> Check:
    dtype = ds["time"].dtype
    if not np.issubdtype(dtype, np.datetime64):
        return fail("time_naive_utc", LEVEL, "time.dtype", str(dtype), "datetime64[ns]")
    if getattr(dtype, "tz", None) is not None:
        return fail("time_naive_utc", LEVEL, "time.dtype", str(dtype), "datetime64[ns] without tz")
    return ok("time_naive_utc", LEVEL)


def _valid_time_consistent(ds: xr.Dataset) -> Check:
    """valid_time == init_time + lead_time, docs/DOMAIN.md §2."""
    if "lead_time" not in ds.coords or "init_time" not in ds.attrs:
        return ok("valid_time_consistent", LEVEL, "no lead_time coordinate; nothing to cross-check")

    init = np.datetime64(str(ds.attrs["init_time"]).rstrip("Z"), "ns")
    leads = np.asarray(ds["lead_time"].values).astype("int64")
    expected = init + leads.astype("timedelta64[h]")
    got = np.asarray(ds["time"].values).astype("datetime64[ns]")
    if not np.array_equal(got, expected):
        bad = int(np.argmax(got != expected))
        return fail(
            "valid_time_consistent",
            LEVEL,
            f"time[{bad}] (lead_time={leads[bad]} h)",
            str(got[bad]),
            str(expected[bad]),
        )
    return ok("valid_time_consistent", LEVEL)
```

```python
"""Уровни «физика» и «здравый смысл» из docs/DATA_CONTRACT.md §4.

Эти две группы ловят то, что не ловится структурой: данные правильной формы
и с правильными подписями, но неверные по существу — °C вместо K, осадки,
вычтенные в обратном порядке, поле, распакованное в константу.
"""

import numpy as np
import xarray as xr

from contracts import canon
from validators.result import Check, fail, ok

MAX_NAN_FRACTION = 0.001
MAX_STEP_JUMP_2T = 15.0
GLOBAL_MEAN_2T = (283.0, 292.0)


def check_physics(ds: xr.Dataset) -> list[Check]:
    checks: list[Check] = []
    for raw_name in ds.data_vars:
        name = str(raw_name)
        values = np.asarray(ds[name].values, dtype="float64")
        checks.append(_nan_fraction(name, values))
        bounds = canon.PHYSICAL_RANGES.get(name)
        if bounds is not None:
            checks.append(_range(name, values, bounds))
    return checks


def check_sanity(ds: xr.Dataset) -> list[Check]:
    checks: list[Check] = []
    for raw_name in ds.data_vars:
        name = str(raw_name)
        checks.append(_not_constant(name, np.asarray(ds[name].values, dtype="float64")))
    if "2t" in ds.data_vars:
        checks.append(_global_mean_2t(ds))
        if ds.sizes.get("time", 1) > 1:
            checks.append(_step_jump_2t(ds))
    return checks


def _finite(values: np.ndarray) -> np.ndarray:
    return values[np.isfinite(values)]


def _nan_fraction(name: str, values: np.ndarray) -> Check:
    fraction = float(np.count_nonzero(~np.isfinite(values)) / values.size)
    if fraction > MAX_NAN_FRACTION:
        return fail("nan_fraction", "physics", name, round(fraction, 6), f"<= {MAX_NAN_FRACTION}")
    return ok("nan_fraction", "physics", f"{name}: {fraction:.6f}")


def _range(name: str, values: np.ndarray, bounds: tuple[float, float]) -> Check:
    finite = _finite(values)
    if finite.size == 0:
        return fail("range", "physics", name, "all values are NaN", bounds)
    lo, hi = float(finite.min()), float(finite.max())
    if lo < bounds[0] or hi > bounds[1]:
        return fail("range", "physics", name, (round(lo, 4), round(hi, 4)), bounds)
    return ok("range", "physics", f"{name}: [{lo:.4g}, {hi:.4g}] within {bounds}")


def _not_constant(name: str, values: np.ndarray) -> Check:
    finite = _finite(values)
    if finite.size and float(finite.min()) == float(finite.max()):
        return fail("not_constant", "sanity", name, f"constant {float(finite.min())}", "a varying field")
    return ok("not_constant", "sanity", name)


def _global_mean_2t(ds: xr.Dataset) -> Check:
    """Средняя приземная температура Земли ~288 K. Около 15 — это °C."""
    weights = np.cos(np.deg2rad(np.asarray(ds["lat"].values)))
    mean = float(ds["2t"].weighted(xr.DataArray(weights, dims="lat")).mean().item())
    if not (GLOBAL_MEAN_2T[0] <= mean <= GLOBAL_MEAN_2T[1]):
        return fail("global_mean_2t", "sanity", "2t", round(mean, 3), GLOBAL_MEAN_2T)
    return ok("global_mean_2t", "sanity", f"{mean:.2f} K")


def _step_jump_2t(ds: xr.Dataset) -> Check:
    """Разница соседних шагов больше 15 K в точке — признак перепутанных шагов."""
    diff = np.abs(np.diff(np.asarray(ds["2t"].values, dtype="float64"), axis=0))
    largest = float(np.nanmax(diff))
    if largest > MAX_STEP_JUMP_2T:
        return fail("step_jump_2t", "sanity", "2t", round(largest, 3), f"<= {MAX_STEP_JUMP_2T} K per 6 h")
    return ok("step_jump_2t", "sanity", f"max jump {largest:.2f} K")
```

```python
"""Валидаторы данных: четыре уровня из docs/DATA_CONTRACT.md §4.

Валидатор ничего не чинит и ничего не публикует. Он выносит вердикт, а решение
«публиковать или нет» принимает вызывающий — pipeline. Отвергнутый срез
не публикуется, предыдущий остаётся актуальным (docs/PIPELINE.md §3).
"""

import json
from collections.abc import Sequence
from pathlib import Path

import xarray as xr

from validators.physics import check_physics, check_sanity
from validators.result import Check, ValidationReport
from validators.semantics import check_semantics
from validators.structure import check_structure

ALL_LEVELS = ("structure", "semantics", "physics", "sanity")

__all__ = [
    "ALL_LEVELS",
    "Check",
    "RejectedError",
    "ValidationReport",
    "raise_if_rejected",
    "validate",
    "write_report",
]


class RejectedError(Exception):
    """Срез не прошёл валидацию и потому не публикуется."""

    def __init__(self, report: ValidationReport) -> None:
        lines = [c.message for c in report.failures()]
        super().__init__("; ".join(lines))
        self.report = report


def validate(
    ds: xr.Dataset,
    *,
    levels: Sequence[str] | None = None,
    expect_static: bool = False,
) -> ValidationReport:
    wanted = tuple(levels) if levels is not None else ALL_LEVELS
    unknown = set(wanted) - set(ALL_LEVELS)
    if unknown:
        raise ValueError(f"levels: got {sorted(unknown)}, expected a subset of {ALL_LEVELS}")

    checks: list[Check] = []
    if "structure" in wanted:
        checks.extend(check_structure(ds, expect_static=expect_static))
    if "semantics" in wanted:
        checks.extend(check_semantics(ds))
    if "physics" in wanted:
        checks.extend(check_physics(ds))
    if "sanity" in wanted:
        checks.extend(check_sanity(ds))
    return ValidationReport(tuple(checks))


def write_report(report: ValidationReport, path: Path) -> Path:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(report.to_dict(), indent=2, ensure_ascii=False), encoding="utf-8")
    tmp.replace(path)
    return path


def raise_if_rejected(report: ValidationReport) -> None:
    if not report.ok:
        raise RejectedError(report)
```

```python
    return 0 if report.ok else 1
```

```python
"""API читает хранилище и кэш. Всё остальное для него не существует."""

import ast
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[1]

FORBIDDEN = {
    "api": {"adapters", "pipeline", "torch", "aurora", "cfgrib", "cdsapi"},
    "storage": {"api", "adapters", "torch", "aurora"},
    "cache": {"api", "pipeline", "torch", "aurora"},
    "contracts": {"api", "storage", "cache", "adapters", "pipeline", "xarray", "torch"},
    "validators": {"api", "storage", "cache", "adapters", "pipeline", "torch"},
}


def _imported_roots(path: Path) -> set[str]:
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    roots: set[str] = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            roots.update(alias.name.split(".")[0] for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.level == 0 and node.module:
            roots.add(node.module.split(".")[0])
    return roots


@pytest.mark.parametrize("package", sorted(FORBIDDEN))
def test_package_does_not_import_what_it_must_not(package: str) -> None:
    banned = FORBIDDEN[package]
    offences: list[str] = []
    for source in (ROOT / package).rglob("*.py"):
        for imported in _imported_roots(source) & banned:
            offences.append(f"{source.relative_to(ROOT)} imports {imported}")
    assert not offences, "\n".join(offences)
```
